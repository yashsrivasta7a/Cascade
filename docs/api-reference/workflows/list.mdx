---
title: 'List Workflows'
api: 'GET /workflows'
description: 'Returns all workflows for the authenticated user'
---

## Response

<ResponseField name="workflows" type="array">
  Array of workflow objects
  
  <Expandable title="Workflow object">
    <ResponseField name="id" type="string">
      Unique workflow identifier
    </ResponseField>
    <ResponseField name="name" type="string">
      Name of the workflow
    </ResponseField>
    <ResponseField name="description" type="string">
      Optional description
    </ResponseField>
    <ResponseField name="version" type="number">
      Current version number
    </ResponseField>
    <ResponseField name="isPublished" type="boolean">
      Whether the workflow is published
    </ResponseField>
    <ResponseField name="createdAt" type="string">
      ISO 8601 creation timestamp
    </ResponseField>
    <ResponseField name="updatedAt" type="string">
      ISO 8601 last update timestamp
    </ResponseField>
    <ResponseField name="_count.executions" type="number">
      Total number of executions
    </ResponseField>
  </Expandable>
</ResponseField>
