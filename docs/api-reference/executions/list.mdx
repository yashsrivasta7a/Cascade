---
title: 'List Executions'
description: 'Retrieve a paginated list of all workflow executions'
---

## Overview

Returns a paginated list of all workflow executions for the authenticated user, including execution statistics.

<Note>
This endpoint is at `/api/executions` (not under `/api/v1`).
</Note>

## Query Parameters

<ParamField query="status" type="string">
  Filter by execution status. Options: `all`, `running`, `completed`, `failed`, `cancelled`
</ParamField>

<ParamField query="search" type="string">
  Search executions by workflow name
</ParamField>

<ParamField query="limit" type="integer" default="20">
  Number of results per page (max 100)
</ParamField>

<ParamField query="offset" type="integer" default="0">
  Number of results to skip for pagination
</ParamField>

## Request

<CodeGroup>
```bash cURL
curl -X GET "https://flowsmiths.vercel.app/api/executions?status=completed&limit=10" \
  -H "Authorization: Bearer YOUR_API_KEY"
```

```javascript JavaScript
const response = await fetch(
  'https://flowsmiths.vercel.app/api/executions?status=completed&limit=10',
  {
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY'
    }
  }
);
const data = await response.json();
```

```python Python
import requests

response = requests.get(
    'https://flowsmiths.vercel.app/api/executions',
    params={'status': 'completed', 'limit': 10},
    headers={'Authorization': 'Bearer YOUR_API_KEY'}
)
data = response.json()
```
</CodeGroup>

## Response

<ResponseField name="executions" type="array">
  List of execution objects
  <Expandable title="Execution Object">
    <ResponseField name="id" type="string">
      Unique execution identifier
    </ResponseField>
    <ResponseField name="type" type="string">
      Type of execution: `workflow` or `quick`
    </ResponseField>
    <ResponseField name="workflowId" type="string">
      ID of the executed workflow
    </ResponseField>
    <ResponseField name="workflowName" type="string">
      Name of the workflow
    </ResponseField>
    <ResponseField name="status" type="string">
      Current status: `running`, `completed`, `failed`, `cancelled`
    </ResponseField>
    <ResponseField name="startedAt" type="string">
      ISO timestamp when execution started
    </ResponseField>
    <ResponseField name="completedAt" type="string">
      ISO timestamp when execution completed (if finished)
    </ResponseField>
    <ResponseField name="duration" type="string">
      Human-readable duration (e.g., "2.5s", "1m 30s")
    </ResponseField>
    <ResponseField name="totalCost" type="number">
      Total credits consumed by this execution
    </ResponseField>
    <ResponseField name="nodeCount" type="integer">
      Number of nodes executed
    </ResponseField>
    <ResponseField name="nodes" type="array">
      Details of individual node executions
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="pagination" type="object">
  Pagination metadata
  <Expandable title="Pagination Object">
    <ResponseField name="total" type="integer">
      Total number of executions
    </ResponseField>
    <ResponseField name="limit" type="integer">
      Results per page
    </ResponseField>
    <ResponseField name="offset" type="integer">
      Current offset
    </ResponseField>
    <ResponseField name="hasMore" type="boolean">
      Whether more results are available
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="stats" type="object">
  Aggregate statistics
  <Expandable title="Stats Object">
    <ResponseField name="totalRuns" type="integer">
      Total number of executions
    </ResponseField>
    <ResponseField name="successRate" type="string">
      Success rate percentage (e.g., "95.5%")
    </ResponseField>
    <ResponseField name="avgDuration" type="string">
      Average execution duration
    </ResponseField>
    <ResponseField name="creditsUsed" type="number">
      Total credits used
    </ResponseField>
  </Expandable>
</ResponseField>
