---
title: 'Get Execution'
api: 'GET /executions/{id}'
description: 'Retrieve detailed information about a specific execution'
---

## Overview

Returns detailed information about a specific workflow execution, including the status of each node.

<ParamField path="id" type="string" required>
  The unique identifier of the execution
</ParamField>

## Request

<CodeGroup>
```bash cURL
curl -X GET https://flowsmiths.vercel.app/api/v1/executions/{id} \
  -H "Authorization: Bearer YOUR_API_KEY"
```

```javascript JavaScript
const response = await fetch(
  'https://flowsmiths.vercel.app/api/v1/executions/{id}',
  {
    headers: {
      'Authorization': 'Bearer YOUR_API_KEY'
    }
  }
);
const data = await response.json();
```

```python Python
import requests

response = requests.get(
    'https://flowsmiths.vercel.app/api/v1/executions/{id}',
    headers={'Authorization': 'Bearer YOUR_API_KEY'}
)
data = response.json()
```
</CodeGroup>

## Response

<ResponseField name="id" type="string">
  Unique execution identifier
</ResponseField>

<ResponseField name="workflowId" type="string">
  ID of the executed workflow
</ResponseField>

<ResponseField name="status" type="string">
  Current status: `pending`, `running`, `completed`, `failed`, `cancelled`
</ResponseField>

<ResponseField name="startedAt" type="string">
  ISO timestamp when execution started
</ResponseField>

<ResponseField name="completedAt" type="string">
  ISO timestamp when execution completed
</ResponseField>

<ResponseField name="estimatedCost" type="number">
  Estimated credit cost before execution
</ResponseField>

<ResponseField name="actualCost" type="number">
  Actual credits consumed (available after completion)
</ResponseField>

<ResponseField name="error" type="string">
  Error message if execution failed
</ResponseField>

<ResponseField name="nodeExecutions" type="array">
  Details for each node execution
  <Expandable title="Node Execution Object">
    <ResponseField name="nodeId" type="string">
      Node identifier within the workflow
    </ResponseField>
    <ResponseField name="nodeType" type="string">
      Type of node (e.g., `text-input`, `image-gen`, `llm`)
    </ResponseField>
    <ResponseField name="nodeLabel" type="string">
      Display name of the node
    </ResponseField>
    <ResponseField name="status" type="string">
      Node status: `pending`, `running`, `completed`, `failed`
    </ResponseField>
    <ResponseField name="startedAt" type="string">
      ISO timestamp when node started
    </ResponseField>
    <ResponseField name="completedAt" type="string">
      ISO timestamp when node completed
    </ResponseField>
    <ResponseField name="providerUsed" type="string">
      AI provider used (e.g., `fal`, `openrouter`)
    </ResponseField>
    <ResponseField name="actualCost" type="number">
      Credits consumed by this node
    </ResponseField>
    <ResponseField name="error" type="string">
      Error message if node failed
    </ResponseField>
    <ResponseField name="output" type="object">
      Node output data (type-specific)
    </ResponseField>
  </Expandable>
</ResponseField>
